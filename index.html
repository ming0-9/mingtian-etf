import { init } from 'klinecharts'; // 1. 初始化图表容器（支持交易所暗色风格） const chart = init('chart-container', { styles: { grid: { show: true }, candle: { type: 'candle_solid' } // 标准蜡烛图 } }); // 2. 主图叠加 BOLL 指标（与交易所一致） chart.createIndicator({ name: 'BOLL', calcParams: [20, 2], // N=20, k=2 paneId: 'candle_pane' // 叠加在主图 K 线区域 }, true); // 3. 创建副图指标：成交量 VOL chart.createIndicator({ name: 'VOL', calcParams: [5, 10] }); // 4. 创建副图指标：MACD chart.createIndicator({ name: 'MACD', calcParams: [12, 26, 9] }); // 5. 创建副图指标：RSI chart.createIndicator({ name: 'RSI', calcParams: [6, 12, 24] }); // 6. 注入行业板块/交易所 K 线数据 (格式为包含 open, high, low, close, volume, timestamp 的数组) const industryData = [ { timestamp: 1700000000000, open: 1200.5, high: 1215.0, low: 1195.2, close: 1210.0, volume: 5400000 }, // ... 更多历史数据 ]; chart.applyNewData(industryData);
