function getActionLogic(c120, cDaily, pct, budget, pePct, pbPct) { // 计算综合估值百分位数（若无PE/PB则默认为50合理区） let avgPct = 50; if (pePct > 0 && pbPct > 0) avgPct = (pePct + pbPct) / 2; else if (pePct > 0) avgPct = pePct; else if (pbPct > 0) avgPct = pbPct; const isLowVal = avgPct <= 20; // 低估机会区 const isHighVal = avgPct >= 80; // 高估危险区 let zone = '', action = '', style = ''; // 【1. 高估危险区逻辑】：防守第一，坚决禁止任何买入/网格加仓 if (isHighVal) { if (c120 >= 100 || cDaily >= 85) { zone = '高估+泡沫见顶'; action = '高估值叠加严重超买，坚决分批止盈/清仓防守'; style = 'bg-rose-900/80 text-rose-300 border-rose-500'; } else { zone = '高估危险区'; action = '历史高位（禁止建仓/加仓），持股者择机逢高减筹'; style = 'bg-rose-950/60 text-rose-400 border-rose-800'; } } // 【2. 低估机会区逻辑】：进攻为主，越跌越买，短线冲高仅做波段 else if (isLowVal) { if (c120 >= 100) { zone = '低估+短线超买'; action = '长线极具性价比！大趋势留底仓，仅波段减筹 20%'; style = 'bg-amber-900/80 text-amber-300 border-amber-500'; } else if (cDaily < 0) { zone = '低估重仓区(建2)'; action = `低估+日线筑底，安全边际极高，补齐至 40% (￥${(budget * 0.4).toLocaleString()})`; style = 'bg-emerald-900/80 text-emerald-300 border-emerald-500'; } else if (c120 < 0 || cDaily < 10) { zone = '低估首仓区(建1)'; action = `低估+动量探底，买入 20% 计划仓位 (￥${(budget * 0.2).toLocaleString()})`; style = 'bg-teal-900/80 text-teal-300 border-teal-500'; } else if (pct <= -3.0) { zone = '低估黄金网格'; action = `低估优质加仓点，追加 10% 资金 (￥${(budget * 0.1).toLocaleString()})`; style = 'bg-cyan-900/80 text-cyan-300 border-cyan-500'; } else { zone = '低估蓄势区'; action = '长线性价比优异，安心持股或分批低吸'; style = 'bg-emerald-950/60 text-emerald-400 border-emerald-800'; } } // 【3. 估值合理区间逻辑】：严格按动量与网格纪律执行 else { if (c120 >= 100) { zone = '减1区域(止盈50%)'; action = '动量超买，减持当前持仓的 50%'; style = 'bg-amber-900/80 text-amber-300 border-amber-500'; } else if (cDaily >= 85) { zone = '减2区域(清仓)'; action = '中期动量见顶，减持 50% 或直接清仓'; style = 'bg-rose-900/80 text-rose-300 border-rose-500'; } else if (cDaily < 0) { zone = '建2区域'; action = `动量探底，买入/补齐至 40% (￥${(budget * 0.4).toLocaleString()})`; style = 'bg-emerald-900/80 text-emerald-300 border-emerald-500'; } else if (c120 < 0 || cDaily < 10) { zone = '建1区域'; action = `动量低位，买入 20% 计划仓位 (￥${(budget * 0.2).toLocaleString()})`; style = 'bg-teal-900/80 text-teal-300 border-teal-500'; } else if (pct <= -3.0) { zone = '网格加仓'; action = `单日跌超 3%，追加 10% 资金 (￥${(budget * 0.1).toLocaleString()})`; style = 'bg-cyan-900/80 text-cyan-300 border-cyan-500'; } else { zone = '合理持有'; action = '观望/按兵不动，等待确切的动量信号'; style = 'bg-slate-800 text-slate-400 border-slate-700'; } } return { zone, action, style }; }
